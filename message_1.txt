        }))

local function LoadConfig()
    if not (readfile and isfile) then return end
    pcall(function()
        if not isfile(CONFIG_FILE) then return end
        local data = HttpService:JSONDecode(readfile(CONFIG_FILE))
        if type(data) ~= "table" then return end
        local function merge(d, s)
            if type(s) ~= "table" then return end
            for k, v in pairs(s) do
                if type(v) == "table" and type(d[k]) == "table" then merge(d[k], v)
                elseif d[k] ~= nil then d[k] = v end
            end
        end
        merge(Settings, data)
    end)
end

local friendCache = {}
local function IsFriendCached(userId)
    if not userId then return false end
    if friendCache[userId] ~= nil then return friendCache[userId] end
    task.spawn(function()
        local f = false
        pcall(function() f = LocalPlayer:IsFriendsWith(userId) end)
        friendCache[userId] = f
    end)
    return false
end

local function IsWhitelisted(player)
    if not player then return false end
    if Settings.Whitelist.Players[player.Name] or Settings.Whitelist.Players[string.lower(player.Name)] then return true end
    if Settings.Whitelist.Team and LocalPlayer.Team and player.Team and LocalPlayer.Team == player.Team then return true end
    if Settings.Whitelist.Friends and IsFriendCached(player.UserId) then return true end
    return false
end

local function IsValidTarget(player)
    if player == LocalPlayer or not player.Character then return false end
    if IsWhitelisted(player) then return false end
    local char = player.Character
    local humanoid = char:FindFirstChild("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not humanoid or humanoid.Health <= 0 or not hrp then return false end
    local be = char:FindFirstChild("BodyEffects")
    if be then
        local ko = be:FindFirstChild("K.O") or be:FindFirstChild("KO")
        if ko and ko.Value == true then return false end
    end
    return true
end

local function GetCharacterAimPart(character, chosenPart)
    if not character then return nil end
    local partName = chosenPart or "HumanoidRootPart"
    if partName == "Random" then
        local parts = {"Head", "UpperTorso", "HumanoidRootPart"}
        partName = parts[math.random(1, #parts)]
    end
    local part = character:FindFirstChild(partName)
    if not part then part = character:FindFirstChild("HumanoidRootPart") or character:FindFirstChild("Head") end
    return part
end

local function GetPredictedPosition(part, usePred, predAmount)
    if not part then return nil end
    local pos = part.Position
    if usePred then
        local hrp = part.Parent and part.Parent:FindFirstChild("HumanoidRootPart")
        local vel = (hrp and (hrp.AssemblyLinearVelocity or hrp.Velocity)) or part.AssemblyLinearVelocity or part.Velocity
        if vel then pos = pos + vel * (predAmount or 0.135) end
    end
    return pos
end

local function GetClosestPlayerForAim()
    local cam = workspace.CurrentCamera or Camera
    if not cam then return nil end
    local closest, shortest = nil, Settings.Aimbot.FOV or 220
    local center = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)
    for _, player in ipairs(Players:GetPlayers()) do
        if IsValidTarget(player) then
            local part = GetCharacterAimPart(player.Character, Settings.Aimbot.AimPart)
            if part then
                local sp, on = cam:WorldToViewportPoint(part.Position)
                if on and sp.Z > 0 then
                    local d = (Vector2.new(sp.X, sp.Y) - center).Magnitude
                    if d <= shortest then shortest = d closest = player end
                end
            end
        end
    end
    return closest
end

        local hrp = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
        if hrp then currentFlyCFrame = hrp.CFrame end
        flyConnection = RunService.Heartbeat:Connect(function(dt)
            if not Settings.Misc.Fly then return end
            local char = LocalPlayer.Character
            local h = char and char:FindFirstChild("HumanoidRootPart")
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            if not h or not hum then return end
            local cam = workspace.CurrentCamera
            if not cam then return end
            if not currentFlyCFrame then currentFlyCFrame = h.CFrame end
            h.AssemblyLinearVelocity = Vector3.zero
            local mv = Vector3.zero
            local cf = cam.CFrame
            if UserInputService:IsKeyDown(Enum.KeyCode.W) then mv = mv + cf.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then mv = mv - cf.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) then mv = mv - cf.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) then mv = mv + cf.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.Space) then mv = mv + Vector3.new(0, 1, 0) end
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then mv = mv - Vector3.new(0, 1, 0) end
            if mv.Magnitude > 0 then
                currentFlyCFrame = currentFlyCFrame + mv.Unit * (Settings.Misc.FlySpeed * (dt or 0.016))
            end
            h.CFrame = CFrame.new(currentFlyCFrame.Position) * cf.Rotation
            h.AssemblyLinearVelocity = Vector3.zero
        end)
        table.insert(_G.CociteConnections, flyConnection)
    else
        if flyConnection then flyConnection:Disconnect() flyConnection = nil end
        currentFlyCFrame = nil
    end
end

local function ToggleSilentLock()
    if not Settings.SilentAim.Enabled then return end
    if lockedSilentTarget then
        lockedSilentTarget = nil
        SilentFloatBtn.Text = "SILENT: ON"
        return
    end
    local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    local best, bestD = nil, Settings.SilentAim.FOV
    for _, plr in ipairs(Players:GetPlayers()) do
        if IsValidTarget(plr) then
            local part = GetCharacterAimPart(plr.Character, Settings.SilentAim.AimPart)
            if part then
                local sp, on = Camera:WorldToViewportPoint(part.Position)

                end
            end
        end
    end
    if best then
        lockedSilentTarget = best
        SilentFloatBtn.Text = "LOCKED: " .. best.Name
        SilentFloatBtn.BackgroundColor3 = Color3.fromRGB(0, 210, 110)
    end
end

-- ===================== GUN PATCH: ONE-SHOT ONLY (NO HEARTBEAT) =====================
local originalVals = {}
local scanning = false

local function OneShotGunPatch()
    if scanning or not getgc then return end
    if not (Settings.RapidFire.Enabled or Settings.NoRecoil.Enabled or Settings.InfiniteAmmo.Enabled) then return end
    scanning = true
    task.spawn(function()
        pcall(function()
            for _, v in pairs(getgc(false)) do -- false = lighter than true
                if typeof(v) == "table" then
                    for _, key in ipairs({"FireRate", "Cooldown", "cooldown", "FireSpeed", "ShootDelay", "Recoil", "Kick", "Ammo", "Clip", "MaxAmmo"}) do
                        local val = rawget(v, key)
                        if type(val) == "number" then
                            if not originalVals[v] then originalVals[v] = {} end
                            if originalVals[v][key] == nil then originalVals[v][key] = val end
                            if key == "Recoil" or key == "Kick" then
                                if Settings.NoRecoil.Enabled then rawset(v, key, 0) end
                            elseif key == "Ammo" or key == "Clip" or key == "MaxAmmo" then
                                if Settings.InfiniteAmmo.Enabled then rawset(v, key, 999) end
                            else
                                if Settings.RapidFire.Enabled and val > 0.01 and val < 5 then
                                    local multi = math.max(1, Settings.RapidFire.Multiplier or 4)
                                    rawset(v, key, math.max(Settings.RapidFire.MinDelay or 0.03, val / multi))
                                end
                            end
                        end
                    end
                end
            end
        end)
        scanning = false
    end)
end

local function RestoreGuns()
    for tbl, keys in pairs(originalVals) do
        for key, orig in pairs(keys) do
            pcall(function() rawset(tbl, key, orig) end)
        end
    end
end

    if hookmetamethod and getnamecallmethod and newcclosure and checkcaller then
        local oldNamecall
        oldNamecall = hookmetamethod(game, "__namecall", newcclosure(function(...)
            local method = getnamecallmethod()
            local args = {...}
            if not checkcaller() and AUTHED and Settings.SilentAim.Enabled and currentSilentAimPos and not isInternalRay then
                if args[1] == workspace and method == "Raycast" and typeof(args[2]) == "Vector3" and typeof(args[3]) == "Vector3" then
                    args[3] = (currentSilentAimPos - args[2]).Unit * 1000
                    return oldNamecall(table.unpack(args))
                elseif method == "FireServer" and tostring(args[1]) == "MainEvent" then
                    local ev = tostring(args[2] or "")
                    if string.find(string.lower(ev), "mousepos") then
                        if args[2] == ev then args[3] = currentSilentAimPos else args[2] = currentSilentAimPos end
                        return oldNamecall(table.unpack(args))
                    end
                end
            end
            return oldNamecall(...)
        end))
    end
end)

    if not AUTHED or not Settings.Aimbot.Enabled or not AimActive then return end
    local cam = workspace.CurrentCamera
    if not cam then return end
    local target = Settings.Aimbot.LockedTarget
    if not target or not IsValidTarget(target) then
        target = GetClosestPlayerForAim()
        Settings.Aimbot.LockedTarget = target
    end
    if target then
        local part = GetCharacterAimPart(target.Character, Settings.Aimbot.AimPart)
        if part then
            local aimPos = GetPredictedPosition(part, Settings.Aimbot.UsePrediction, Settings.Aimbot.Prediction)
            if aimPos then
                local targetCF = CFrame.lookAt(cam.CFrame.Position, aimPos)
                if Settings.Aimbot.Smooth then
                    cam.CFrame = cam.CFrame:Lerp(targetCF, math.clamp(Settings.Aimbot.Smoothness or 0.35, 0.05, 1))
                else
                    cam.CFrame = targetCF
                end
            end
        end
    end
end)
