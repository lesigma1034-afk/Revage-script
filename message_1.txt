local function IsValidTarget(player)
    if player == LocalPlayer or not player.Character then return false end
    local char = player.Character
    local humanoid = char:FindFirstChild("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")
    
    if not humanoid or humanoid.Health <= 0 or not hrp then return false end
    
    local bodyEffects = char:FindFirstChild("BodyEffects")
    if bodyEffects then
        local ko = bodyEffects:FindFirstChild("K.O") or bodyEffects:FindFirstChild("KO")
        if ko and ko.Value == true then
            return false
        end
    end
    
    return true
end

local function GetClosestSilentTarget()
    local closestP = nil
    local shortestDist = Settings.SilentAim.FOV
    local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    
    for _, player in ipairs(Players:GetPlayers()) do
        if IsValidTarget(player) then
            local char = player.Character
            local part = char:FindFirstChild(Settings.SilentAim.HitPart) or char:FindFirstChild("HumanoidRootPart")
            if part then
                local pos, onScreen = Camera:WorldToViewportPoint(part.Position)
                if onScreen then
                    local magnitude = (Vector2.new(pos.X, pos.Y) - center).Magnitude
                    if magnitude < shortestDist then
                        shortestDist = magnitude
                        closestP = player
                    end
                end
            end
        end
    end
    return closestP
end

-- Функция создания белого трейсера как в видео
local function ShowTracer(targetPart)
    if not Settings.SilentAim.Tracers or not LocalPlayer.Character then return end
    local tool = LocalPlayer.Character:FindFirstChildOfClass("Tool")
    local originPart = tool and tool:FindFirstChild("Handle") or LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not originPart then return end
    
    local line = Drawing.new("Line")
    line.Thickness = 1.5
    line.Color = Color3.fromRGB(255, 255, 255)
    line.Transparency = 1
    
    local connection
    connection = RunService.RenderStepped:Connect(function()
        if originPart and targetPart then
            local p1, on1 = Camera:WorldToViewportPoint(originPart.Position)
            local p2, on2 = Camera:WorldToViewportPoint(targetPart.Position)
            if on1 and on2 then
                line.From = Vector2.new(p1.X, p1.Y)
                line.To = Vector2.new(p2.X, p2.Y)
                line.Visible = true
            else
                line.Visible = false
            end
        else
            line.Visible = false
        end
    end)
    
    task.delay(0.06, function()
        if connection then connection:Disconnect() end
        line:Remove()
    end)
end

pcall(function()
    local mt = getrawmetatable(game)
    setreadonly(mt, false)
    local oldNamecall = mt.__namecall
    
    mt.__namecall = newcclosure(function(self, ...)
        local method = getnamecallmethod()
        local args = {...}
        if Settings.SilentAim.Enabled and (method == "FindPartOnRay" or method == "FindPartOnRayWithIgnoreList") then
            local target = GetClosestSilentTarget()
            if target and target.Character then
                local part = target.Character:FindFirstChild(Settings.SilentAim.HitPart) or target.Character:FindFirstChild("HumanoidRootPart")
                if part and args[1] then
                    ShowTracer(part)
                    local origin = args[1].Origin
                    local direction = (part.Position - origin).Unit * 1000
                    args[1] = Ray.new(origin, direction)
                    return oldNamecall(self, unpack(args))
                end
            end
        elseif Settings.SilentAim.Enabled and method == "Raycast" then
            local target = GetClosestSilentTarget()
            if target and target.Character then
                local part = target.Character:FindFirstChild(Settings.SilentAim.HitPart) or target.Character:FindFirstChild("HumanoidRootPart")
                if part and args[1] and args[2] then
                    ShowTracer(part)
                    local origin = args[1]
                    local direction = (part.Position - origin).Unit * 1000
                    args[2] = direction
                    return oldNamecall(self, unpack(args))
                end
            end
        end
        return oldNamecall(self, ...)
    end)
end)

local function RemoveHighlight(player)

local function ClearESP()

        for _, obj in pairs(objList) do
            if obj then obj:Remove() end
        end
    end
    Settings.ESP.ActiveDrawings = {}

    for player, _ in pairs(Settings.ESP.ActiveHighlights) do
        RemoveHighlight(player)
    end
end

Players.PlayerRemoving:Connect(function(player)
    if Settings.ESP.ActiveDrawings[player] then
        for _, obj in pairs(Settings.ESP.ActiveDrawings[player]) do
            if obj then obj:Remove() end
        end
        Settings.ESP.ActiveDrawings[player] = nil
    end
    RemoveHighlight(player)
end)

RunService.RenderStepped:Connect(function()
    for _, player in ipairs(Players:GetPlayers()) do
        if Settings.ESP.Enabled and IsValidTarget(player) then
            local char = player.Character
            local hrp = char:FindFirstChild("HumanoidRootPart")
            local pos, onScreen = Camera:WorldToViewportPoint(hrp.Position)
            
            if not Settings.ESP.ActiveDrawings[player] then
                local box = Drawing.new("Square")
                box.Visible = false
                box.Color = Color3.fromRGB(180, 50, 255)
                box.Thickness = 1.5
                box.Filled = false
                
                local nameText = Drawing.new("Text")
                nameText.Visible = false
                nameText.Color = Color3.fromRGB(255, 255, 255)
                nameText.Size = 13
                nameText.Center = true
                nameText.Outline = true
                
                Settings.ESP.ActiveDrawings[player] = {Box = box, Name = nameText}
            end
            
            local visuals = Settings.ESP.ActiveDrawings[player]
            local box, nameText = visuals.Box, visuals.Name
            
            if onScreen then
                local scaleFactor = 300 / pos.Z
                local size = Vector2.new(2, 3) * scaleFactor
                box.Size = size
                box.Position = Vector2.new(pos.X - size.X / 2, pos.Y - size.Y / 2)
                box.Visible = Settings.ESP.Boxes
                
                nameText.Text = player.Name
                nameText.Position = Vector2.new(pos.X, pos.Y - size.Y / 2 - 18)
                nameText.Visible = Settings.ESP.Names
            else
                box.Visible = false
                nameText.Visible = false
            end

            if Settings.ESP.Chams then
                if not Settings.ESP.ActiveHighlights[player] or Settings.ESP.ActiveHighlights[player].Parent ~= char then
                    RemoveHighlight(player)
                    local hl = Instance.new("Highlight")
                    hl.Name = "CociteChams"
                    hl.FillColor = Color3.fromRGB(150, 40, 255)
                    hl.OutlineColor = Color3.fromRGB(255, 255, 255)
                    hl.FillTransparency = 0.4
                    hl.OutlineTransparency = 0
                    hl.Adornee = char
                    hl.Parent = char
                    Settings.ESP.ActiveHighlights[player] = hl
                else
                    Settings.ESP.ActiveHighlights[player].Enabled = true
                end
            else
                RemoveHighlight(player)
            end
        else
            if Settings.ESP.ActiveDrawings[player] then
                Settings.ESP.ActiveDrawings[player].Box.Visible = false
                Settings.ESP.ActiveDrawings[player].Name.Visible = false
            end
            RemoveHighlight(player)
        end
    end
end)

-- Инициализация кнопок ESP в меню
CreateToggle(ESPPage, "Master ESP", false, function(state)
    Settings.ESP.Enabled = state
    if not state then
        ClearESP()
    end
end)

CreateToggle(ESPPage, "Box ESP", true, function(state)
    Settings.ESP.Boxes = state
end)

CreateToggle(ESPPage, "Chams ESP", true, function(state)
    Settings.ESP.Chams = state
end)

CreateToggle(ESPPage, "Name ESP", true, function(state)
    Settings.ESP.Names = state
end)
