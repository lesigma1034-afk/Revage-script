local function CreateMobileButton(name, defaultState, callback)
    local Btn = Instance.new("TextButton")
    Btn.Parent = MainFrame
    Btn.BackgroundColor3 = defaultState and Color3.fromRGB(0, 150, 75) or Color3.fromRGB(40, 40, 40)
    Btn.Size = UDim2.new(1, -10, 0, 35)
    Btn.Font = Enum.Font.SourceSansBold
    Btn.Text = name .. ": " .. (defaultState and "ON" or "OFF")
    Btn.TextColor3 = Color3.fromRGB(255, 255, 255)
    Btn.TextSize = 14

    Btn.Activated:Connect(function()
        defaultState = not defaultState
        Btn.Text = name .. ": " .. (defaultState and "ON" or "OFF")
        Btn.BackgroundColor3 = defaultState and Color3.fromRGB(0, 150, 75) or Color3.fromRGB(40, 40, 40)
        callback(defaultState)
    end)
end

-- Обработка нажатия на кнопку открытия (дублируем Activated и MouseButton1Click для надежности)
local function ToggleWindow()
    MainFrame.Visible = not MainFrame.Visible
end

ToggleButton.Activated:Connect(ToggleWindow)
ToggleButton.MouseButton1Click:Connect(ToggleWindow)

CreateMobileButton("Inventory Sorter", false, function(state)
    if state and KeanW.InventorySorter.Enabled then
        local backpack = LocalPlayer.Backpack
        for _, x in ipairs(KeanW.InventorySorter.List) do
            for _, tool in ipairs(backpack:GetChildren()) do
                if tool:IsA("Tool") and string.find(string.lower(tool.Name), string.lower(x)) then
                    tool.Parent = LocalPlayer.Character
                    task.wait(0.05)
                    tool.Parent = backpack
                end
            end
        end
    end
end)

-- Мобильный ESP
local espf = Instance.new("Folder")
pcall(function()
    espf.Parent = gethui() or game:GetService("CoreGui")
end)
if not espf.Parent then
    espf.Parent = LocalPlayer:WaitForChild("PlayerGui")
end
espf.Name = "MobileESP"

local function addEsp(player)
    if player == LocalPlayer then return end
    local bbg = Instance.new("BillboardGui", espf)
    bbg.Name = player.Name
    bbg.AlwaysOnTop = true
    bbg.Size = UDim2.new(4, 0, 5.4, 0)
    
    local namelabel = Instance.new("TextLabel", bbg)
    namelabel.BackgroundTransparency = 1
    namelabel.Size = UDim2.new(1, 0, 0, 20)
    namelabel.TextColor3 = Color3.fromRGB(255, 255, 255)
    namelabel.TextStrokeTransparency = 0
    namelabel.Text = player.Name
    namelabel.TextSize = 12

    RS.Heartbeat:Connect(function()
        if KeanW.Esp.Enabled and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
            bbg.Adornee = player.Character.HumanoidRootPart
            bbg.Enabled = true
        else
            bbg.Enabled = false
        end
    end)
end

for _, p in pairs(Players:GetPlayers()) do
    addEsp(p)
end
Players.PlayerAdded:Connect(addEsp)
