local function IsWhitelisted(player)
    for _, name in ipairs(Settings.Whitelist) do
        if string.lower(name) == string.lower(player.Name) then
            return true
        end
    end
    return false
end

local function IsValidTarget(player)
    if player == LocalPlayer or not player.Character then return false end
    if IsWhitelisted(player) then return false end
    
    local char = player.Character
    local humanoid = char:FindFirstChild("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")
    
    if not humanoid or humanoid.Health <= 0 or not hrp then return false end
    
    local bodyEffects = char:FindFirstChild("BodyEffects")
    if bodyEffects then
        local ko = bodyEffects:FindFirstChild("K.O") or bodyEffects:FindFirstChild("KO")
        if ko and ko.Value == true then
            return false
        end
    end
    
    return true
end

local function CreateToggle(parent, text, defaultState, callback)
    local btn = Instance.new("TextButton", parent)
    btn.BackgroundColor3 = defaultState and Color3.fromRGB(110, 25, 200) or Color3.fromRGB(20, 15, 32)
    btn.Size = UDim2.new(1, 0, 0, 38)
    btn.Font = Enum.Font.GothamSemibold
    btn.Text = text .. (defaultState and ": ON" or ": OFF")
    btn.TextColor3 = defaultState and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(190, 180, 210)
    btn.TextSize = 12
    
    Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 8)
    local stroke = Instance.new("UIStroke", btn)
    stroke.Color = Color3.fromRGB(90, 40, 160)
    stroke.Thickness = 1.2
    
    local active = defaultState
    btn.MouseButton1Click:Connect(function()
        active = not active
        if active then
            btn.BackgroundColor3 = Color3.fromRGB(110, 25, 200)
            btn.TextColor3 = Color3.fromRGB(255, 255, 255)
            btn.Text = text .. ": ON"
        else
            btn.BackgroundColor3 = Color3.fromRGB(20, 15, 32)
            btn.TextColor3 = Color3.fromRGB(190, 180, 210)
            btn.Text = text .. ": OFF"
        end
        callback(active)
    end)
    return btn
end

local function RefreshWlUI()
    for _, child in ipairs(WlListScroll:GetChildren()) do
        if child:IsA("TextButton") then child:Destroy() end
    end
    for _, name in ipairs(Settings.Whitelist) do
        local item = Instance.new("TextButton", WlListScroll)
        item.BackgroundColor3 = Color3.fromRGB(26, 20, 40)
        item.Size = UDim2.new(1, 0, 0, 24)
        item.Font = Enum.Font.GothamSemibold
        item.Text = "  " .. name .. "  [REMOVE]"
        item.TextColor3 = Color3.fromRGB(255, 100, 100)
        item.TextSize = 11
        item.TextXAlignment = Enum.TextXAlignment.Left
        Instance.new("UICorner", item).CornerRadius = UDim.new(0, 4)
        item.MouseButton1Click:Connect(function()
            for i, n in ipairs(Settings.Whitelist) do
                if n == name then table.remove(Settings.Whitelist, i) break end
            end
            RefreshWlUI()
        end)
    end
    WlListScroll.CanvasSize = UDim2.new(0, 0, 0, #Settings.Whitelist * 28)
end

local function GetClosestToCenter()
    local closestP = nil
    local shortestDist = Settings.Aimbot.FOV
    local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    
    for _, player in ipairs(Players:GetPlayers()) do
        if IsValidTarget(player) then
            local aimPos = GetTargetAimPosition(player.Character)
            if aimPos then
                local pos, onScreen = Camera:WorldToViewportPoint(aimPos)
                if onScreen then
                    local magnitude = (Vector2.new(pos.X, pos.Y) - center).Magnitude
                    if magnitude < shortestDist then
                        shortestDist = magnitude
                        closestP = player
                    end
                end
            end
        end
    end
    return closestP
end

-- // SILENT AIM RAYCAST HOOK
local OldNamecall
OldNamecall = hookmetamethod(game, "__namecall", newcclosure(function(self, ...)
    local method = getnamecallmethod()
    local args = {...}

    if Settings.Aimbot.SilentAim and AimActive and Settings.Aimbot.Enabled then
        if self == workspace or self == Camera then
            if method == "FindPartOnRayWithIgnoreList" or method == "FindPartOnRay" or method == "Raycast" then
                local target = Settings.Aimbot.LockedTarget
                if not target or not IsValidTarget(target) then
                    target = GetClosestToCenter()
                    Settings.Aimbot.LockedTarget = target
                end

                if target and target.Character then
                    local aimPart = target.Character:FindFirstChild("Head") or target.Character:FindFirstChild("HumanoidRootPart")
                    if aimPart then
                        local origin
                        local oldRay = args[1]

                        if typeof(oldRay) == "Ray" then
                            origin = oldRay.Origin
                        elseif typeof(oldRay) == "Vector3" and method == "Raycast" then
                            origin = oldRay
                        end

                        if origin then
                            local newDir = (aimPart.Position - origin).Unit
                            if method == "FindPartOnRayWithIgnoreList" or method == "FindPartOnRay" then
                                args[1] = Ray.new(origin, newDir * 1000)
                            elseif method == "Raycast" then
                                args[2] = newDir * 1000
                            end
                        end
                    end
                end
            end
        end
    end

    return OldNamecall(self, unpack(args))
end))

-- // BULLET TP (Hit TP) - hook MainEvent FireServer + GripPos redirect
local function GetBulletTarget()
    return GetClosestToCenter()
end

local function GetBulletPart(character)
    return character:FindFirstChild("Head") or character:FindFirstChild("HumanoidRootPart")
end

-- Layer 1: MainEvent hook
local MainEvent = ReplicatedStorage:FindFirstChild("MainEvent")
local BulletHookOK = false

if MainEvent then
    pcall(function()
        local oldFireServer = MainEvent.FireServer
        MainEvent.FireServer = newcclosure(function(self, ...)
            if not checkcaller() and BulletTP.Enabled and self == MainEvent then
                local args = {...}
                if args[1] == "UpdateMousePos" or args[1] == "MOUSE" then
                    local target = GetBulletTarget()
                    if target and target.Character then
                        local part = GetBulletPart(target.Character)
                        if part then
                            local pred = part.Position + Vector3.new(
                                part.Velocity.X * BulletTP.Prediction.X,
                                part.Velocity.Y * BulletTP.Prediction.Y,
                                part.Velocity.Z * BulletTP.Prediction.Z
                            )
                            args[2] = pred
                            return oldFireServer(self, unpack(args))
                        end
                    end
                end
            end
            return oldFireServer(self, ...)
        end)
        BulletHookOK = true
    end)
end

-- Layer 2: Tool GripPos redirect on activate
local function HookCharacterGrip(character)
    if not character then return end
    character.ChildAdded:Connect(function(child)
        if child:IsA("Tool") then
            local oldGrip = child.GripPos
            local conn
            conn = child.Activated:Connect(function()
                if not BulletTP.Enabled then return end
                local target = GetBulletTarget()
                if target and target.Character then
                    local myHRP = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                    local part = GetBulletPart(target.Character)
                    if myHRP and part then
                        local objSpace = myHRP.CFrame:PointToObjectSpace(part.Position)
                        oldGrip = child.GripPos
                        child.GripPos = Vector3.new(-objSpace.Z, 0, objSpace.X)
                        task.wait(0.3)
                        pcall(function() child.GripPos = oldGrip end)
                    end
                end
            end)
            child.AncestryChanged:Connect(function()
                if not child.Parent and conn then conn:Disconnect() end
            end)
        end
    end)
end

pcall(function() HookCharacterGrip(LocalPlayer.Character) end)
LocalPlayer.CharacterAdded:Connect(function(char)
    task.wait(0.5)
    HookCharacterGrip(char)
end)

local function RemoveHighlight(player)
    if Settings.ESP.ActiveHighlights[player] then
        Settings.ESP.ActiveHighlights[player]:Destroy()
        Settings.ESP.ActiveHighlights[player] = nil
    end
end

local function ClearESP()
    for _, objList in pairs(Settings.ESP.ActiveDrawings) do
        for _, obj in pairs(objList) do
            if obj then obj:Remove() end
        end
    end
    Settings.ESP.ActiveDrawings = {}

    for player, _ in pairs(Settings.ESP.ActiveHighlights) do
        RemoveHighlight(player)
    end
end

Players.PlayerRemoving:Connect(function(player)
    if Settings.ESP.ActiveDrawings[player] then
        for _, obj in pairs(Settings.ESP.ActiveDrawings[player]) do
            if obj then obj:Remove() end
        end
        Settings.ESP.ActiveDrawings[player] = nil
    end
    RemoveHighlight(player)
end)

RunService.RenderStepped:Connect(function()
    for _, player in ipairs(Players:GetPlayers()) do
        if Settings.ESP.Enabled and IsValidTarget(player) then
            local char = player.Character
            local hrp = char:FindFirstChild("HumanoidRootPart")
            local pos, onScreen = Camera:WorldToViewportPoint(hrp.Position)
            
            if not Settings.ESP.ActiveDrawings[player] then
                local box = Drawing.new("Square")
                box.Visible = false
                box.Color = Color3.fromRGB(180, 50, 255)
                box.Thickness = 1.5
                box.Filled = false
                
                local nameText = Drawing.new("Text")
                nameText.Visible = false
                nameText.Color = Color3.fromRGB(255, 255, 255)
                nameText.Size = 13
                nameText.Center = true
                nameText.Outline = true
                
                Settings.ESP.ActiveDrawings[player] = {Box = box, Name = nameText}
            end
            
            local visuals = Settings.ESP.ActiveDrawings[player]
            local box, nameText = visuals.Box, visuals.Name
            
            if onScreen then
                local scaleFactor = 300 / pos.Z
                local size = Vector2.new(2, 3) * scaleFactor
                box.Size = size
                box.Position = Vector2.new(pos.X - size.X / 2, pos.Y - size.Y / 2)
                box.Visible = Settings.ESP.Boxes
                
                nameText.Text = player.Name
                nameText.Position = Vector2.new(pos.X, pos.Y - size.Y / 2 - 18)
                nameText.Visible = Settings.ESP.Names
            else
                box.Visible = false
                nameText.Visible = false
            end

            if Settings.ESP.Chams then
                if not Settings.ESP.ActiveHighlights[player] or Settings.ESP.ActiveHighlights[player].Parent ~= char then
                    RemoveHighlight(player)
                    local hl = Instance.new("Highlight")
                    hl.Name = "CociteChams"
                    hl.FillColor = Color3.fromRGB(150, 40, 255)
                    hl.OutlineColor = Color3.fromRGB(255, 255, 255)
                    hl.FillTransparency = 0.4
                    hl.OutlineTransparency = 0
                    hl.Adornee = char
                    hl.Parent = char
                    Settings.ESP.ActiveHighlights[player] = hl
                else
                    Settings.ESP.ActiveHighlights[player].Enabled = true
                end
            else
                RemoveHighlight(player)
            end
        else
            if Settings.ESP.ActiveDrawings[player] then
                Settings.ESP.ActiveDrawings[player].Box.Visible = false
                Settings.ESP.ActiveDrawings[player].Name.Visible = false
            end
            RemoveHighlight(player)
        end
    end
end)
