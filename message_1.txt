local function IsWhitelisted(player)
    if not player then return false end
    if Settings.Whitelist.Players[player.Name] 
       or Settings.Whitelist.Players[player.UserId] 
       or Settings.Whitelist.Players[tostring(player.UserId)] 
       or Settings.Whitelist.Players[string.lower(player.Name)] then
        return true
    end
    if Settings.Whitelist.Team and LocalPlayer.Team and player.Team and LocalPlayer.Team == player.Team then
        return true
    end
    if Settings.Whitelist.Friends then
        local isFriend = false
        pcall(function() isFriend = LocalPlayer:IsFriendsWith(player.UserId) end)
        if isFriend then return true end
    end
    return false
end

local function IsValidTarget(player)
    if player == LocalPlayer or not player.Character then return false end
    if IsWhitelisted(player) then return false end
    local char = player.Character
    local humanoid = char:FindFirstChild("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")
    
    if not humanoid or humanoid.Health <= 0 or not hrp then return false end
    
    local bodyEffects = char:FindFirstChild("BodyEffects")
    if bodyEffects then
        local ko = bodyEffects:FindFirstChild("K.O") or bodyEffects:FindFirstChild("KO")
        if ko and ko.Value == true then
            return false
        end
    end
    
    return true
end

local function CreateToggle(parent, text, defaultState, callback)
    local btn = Instance.new("TextButton", parent)
    btn.BackgroundColor3 = defaultState and Color3.fromRGB(110, 25, 200) or Color3.fromRGB(20, 15, 32)
    btn.Size = UDim2.new(1, 0, 0, 34)
    btn.Font = Enum.Font.GothamSemibold
    btn.Text = text .. (defaultState and ": ON" or ": OFF")
    btn.TextColor3 = defaultState and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(190, 180, 210)
    btn.TextSize = 11

    Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 8)
    local stroke = Instance.new("UIStroke", btn)
    stroke.Color = Color3.fromRGB(90, 40, 160)
    stroke.Thickness = 1.2

    local active = defaultState
    btn.MouseButton1Click:Connect(function()
        active = not active
        if active then
            btn.BackgroundColor3 = Color3.fromRGB(110, 25, 200)
            btn.TextColor3 = Color3.fromRGB(255, 255, 255)
            btn.Text = text .. ": ON"
        else
            btn.BackgroundColor3 = Color3.fromRGB(20, 15, 32)
            btn.TextColor3 = Color3.fromRGB(190, 180, 210)
            btn.Text = text .. ": OFF"
        end
        callback(active)
    end)

    return {
        Button = btn,
        Set = function(self, val)
            active = val
            btn.BackgroundColor3 = active and Color3.fromRGB(110, 25, 200) or Color3.fromRGB(20, 15, 32)
            btn.TextColor3 = active and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(190, 180, 210)
            btn.Text = text .. (active and ": ON" or ": OFF")
            callback(active)
        end
    }
end

local function CreateCycle(parent, label, options, defaultIndex, callback)
    local idx = defaultIndex or 1
    local btn = Instance.new("TextButton", parent)
    btn.BackgroundColor3 = Color3.fromRGB(20, 15, 32)
    btn.Size = UDim2.new(1, 0, 0, 34)
    btn.Font = Enum.Font.GothamSemibold
    btn.Text = label .. ": [" .. tostring(options[idx]) .. "]"
    btn.TextColor3 = Color3.fromRGB(210, 190, 255)
    btn.TextSize = 11

    Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 8)
    local stroke = Instance.new("UIStroke", btn)
    stroke.Color = Color3.fromRGB(90, 40, 160)
    stroke.Thickness = 1.2

    btn.MouseButton1Click:Connect(function()
        idx = idx + 1
        if idx > #options then idx = 1 end
        btn.Text = label .. ": [" .. tostring(options[idx]) .. "]"
        callback(options[idx])
    end)

    return {
        Button = btn,
        Get = function() return options[idx] end,
        Set = function(self, val)
            for i, opt in ipairs(options) do
                if opt == val then
                    idx = i
                    btn.Text = label .. ": [" .. tostring(options[idx]) .. "]"
                    callback(options[idx])
                    break
                end
            end
        end
    }
end

-- Core Targeting Logic
local function GetCharacterAimPart(character, chosenPart)
    if not character then return nil end
    local partName = chosenPart or "Head"
    if partName == "Random" then
        local parts = {"Head", "UpperTorso", "HumanoidRootPart"}
        partName = parts[math.random(1, #parts)]
    end
    local part = character:FindFirstChild(partName)
    if not part and (partName == "Head" or partName == "UpperTorso") then
        part = character:FindFirstChild("Torso") or character:FindFirstChild("HumanoidRootPart")
    end
    return part or character:FindFirstChild("HumanoidRootPart") or character:FindFirstChild("Head")
end

local function GetPredictedPosition(part, usePred, predAmount)
    if not part then return nil end
    local pos = part.Position
    if usePred then
        local vel = part.AssemblyLinearVelocity or part.Velocity
        if vel then
            pos = pos + (vel * (predAmount or 0.135))
        end
    end
    return pos
end

local function IsWallCheckVisible(part, targetChar)
    if not Settings.SilentAim.WallCheck then return true end
    local localChar = LocalPlayer.Character
    if not localChar then return false end

    local origin = Camera.CFrame.Position
    local dest = part.Position
    local dir = dest - origin

    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = {localChar, Camera}
    params.IgnoreWater = true

    local hit = workspace:Raycast(origin, dir, params)
    if not hit or hit.Instance:IsDescendantOf(targetChar) then
        return true
    end
    return false
end

local function GetClosestToCenter()
    local closestP = nil
    local shortestDist = Settings.Aimbot.FOV
    local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)

    for _, player in ipairs(Players:GetPlayers()) do
        if IsValidTarget(player) then
            local part = GetCharacterAimPart(player.Character, Settings.Aimbot.AimPart)
            if part then
                local aimPos = GetPredictedPosition(part, Settings.Aimbot.UsePrediction, Settings.Aimbot.Prediction)
                local pos, onScreen = Camera:WorldToViewportPoint(aimPos)
                if onScreen and pos.Z > 0 then
                    local magnitude = (Vector2.new(pos.X, pos.Y) - center).Magnitude
                    if magnitude < shortestDist then
                        shortestDist = magnitude
                        closestP = player
                    end
                end
            end
        end
    end
    return closestP
end

local function GetSilentAimTarget()
    if not Settings.SilentAim.Enabled then return nil, nil end
    if Settings.SilentAim.HitChance < 100 and math.random(1, 100) > Settings.SilentAim.HitChance then
        return nil, nil
    end

    local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    local bestTarget = nil
    local bestPart = nil
    local shortestDist = Settings.SilentAim.FOV

    for _, player in ipairs(Players:GetPlayers()) do
        if IsValidTarget(player) then
            local part = GetCharacterAimPart(player.Character, Settings.SilentAim.AimPart)
            if part and IsWallCheckVisible(part, player.Character) then
                local screenPos, onScreen = Camera:WorldToViewportPoint(part.Position)
                if onScreen and screenPos.Z > 0 then
                    local dist = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
                    if dist <= shortestDist then
                        shortestDist = dist
                        bestTarget = player
                        bestPart = part
                    end
                end
            end
        end
    end
    return bestTarget, bestPart
end

local function RedirectRay(origin, direction, targetPart)
    local targetPos = GetPredictedPosition(targetPart, Settings.SilentAim.UsePrediction, Settings.SilentAim.Prediction)
    local mag = direction.Magnitude
    if mag < 1 then mag = 1000 end
    return (targetPos - origin).Unit * mag
end

-- Hook 1: __index for Mouse.Hit & Mouse.Target (Universal Da Hood gun support)
pcall(function()

        local oldIndex
        oldIndex = hookmetamethod(game, "__index", newcclosure(function(self, key)
            if not checkcaller() and Settings.SilentAim.Enabled then
                if self == Mouse or (typeof(self) == "Instance" and self:IsA("Mouse")) then
                    if key == "Hit" or key == "hit" then
                        local _, targetPart = GetSilentAimTarget()
                        if targetPart then
                            local predictedPos = GetPredictedPosition(targetPart, Settings.SilentAim.UsePrediction, Settings.SilentAim.Prediction)
                            return CFrame.new(predictedPos)
                        end
                    elseif key == "Target" or key == "target" then
                        local _, targetPart = GetSilentAimTarget()
                        if targetPart then
                            return targetPart
                        end
                    end
                end
            end
            return oldIndex(self, key)
        end))
    end
end)

-- Hook 2: __namecall for Da Hood MainEvent & Raycast
pcall(function()
    if hookmetamethod and getnamecallmethod and newcclosure and checkcaller then
        local oldNamecall
        oldNamecall = hookmetamethod(game, "__namecall", newcclosure(function(...)
            local self = ...
            local method = getnamecallmethod()
            local args = {...}

            if not checkcaller() and Settings.SilentAim.Enabled then
                if method == "FireServer" and tostring(self) == "MainEvent" then
                    local eventArg = tostring(args[2] or args[1] or "")
                    if eventArg == "UpdateMousePos" or eventArg == "UpdateMousePosI" or string.find(string.lower(eventArg), "mousepos") then
                        local _, targetPart = GetSilentAimTarget()
                        if targetPart then
                            local predictedPos = GetPredictedPosition(targetPart, Settings.SilentAim.UsePrediction, Settings.SilentAim.Prediction)
                            if args[2] == eventArg then
                                args[3] = predictedPos
                            else
                                args[2] = predictedPos
                            end
                            return oldNamecall(table.unpack(args))
                        end
                    end
                elseif self == workspace then
                    if method == "Raycast" and typeof(args[2]) == "Vector3" and typeof(args[3]) == "Vector3" then
                        local _, targetPart = GetSilentAimTarget()
                        if targetPart then
                            if Settings.SilentAim.TPBullet then
                                args[2] = targetPart.Position + Vector3.new(0, 0.5, 0)
                                args[3] = Vector3.new(0, -1, 0)
                            else
                                args[3] = RedirectRay(args[2], args[3], targetPart)
                            end
                            return oldNamecall(table.unpack(args))
                        end
                    elseif method == "FindPartOnRay" or method == "FindPartOnRayWithIgnoreList" or method == "FindPartOnRayWithWhitelist" then
                        local ray = args[2]
                        if typeof(ray) == "Ray" then
                            local _, targetPart = GetSilentAimTarget()
                            if targetPart then
                                if Settings.SilentAim.TPBullet then
                                    args[2] = Ray.new(targetPart.Position + Vector3.new(0, 0.5, 0), Vector3.new(0, -1, 0))
                                else
                                    local newDir = RedirectRay(ray.Origin, ray.Direction, targetPart)
                                    args[2] = Ray.new(ray.Origin, newDir)
                                end
                                return oldNamecall(table.unpack(args))
                            end
                        end
                    end
                end
            end

            return oldNamecall(...)
        end))
    end
end)

pcall(function()
    if hookfunction and workspace.Raycast and checkcaller and newcclosure then
        local oldRaycast
        oldRaycast = hookfunction(workspace.Raycast, newcclosure(function(self, origin, direction, ...)
            if not checkcaller() and Settings.SilentAim.Enabled and typeof(origin) == "Vector3" and typeof(direction) == "Vector3" then
                local _, targetPart = GetSilentAimTarget()
                if targetPart then
                    if Settings.SilentAim.TPBullet then
                        origin = targetPart.Position + Vector3.new(0, 0.5, 0)
                        direction = Vector3.new(0, -1, 0)
                    else
                        direction = RedirectRay(origin, direction, targetPart)
                    end
                end
            end
            return oldRaycast(self, origin, direction, ...)
        end))
    end
end)

local function RemoveHighlight(player)
    if Settings.ESP.ActiveHighlights[player] then

        Settings.ESP.ActiveHighlights[player] = nil

local function ClearESP()
    for _, objList in pairs(Settings.ESP.ActiveDrawings) do
        for _, obj in pairs(objList) do
            if obj then obj:Remove() end
        end
    end
    Settings.ESP.ActiveDrawings = {}

    for player, _ in pairs(Settings.ESP.ActiveHighlights) do
        RemoveHighlight(player)
    end
end
