local function IsWhitelisted(player)
    for _, name in ipairs(Settings.Whitelist) do
        local low = string.lower(name)
        if low == string.lower(player.Name) or low == string.lower(player.DisplayName) then
            return true
        end
    end
    return false
end

local function IsValidTarget(player)
    if player == LocalPlayer or not player.Character then return false end
    if IsWhitelisted(player) then return false end

    local char = player.Character
    local humanoid = char:FindFirstChild("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")

    if not humanoid or humanoid.Health <= 0 or not hrp then return false end

    local bodyEffects = char:FindFirstChild("BodyEffects")
    if bodyEffects then
        local ko = bodyEffects:FindFirstChild("K.O") or bodyEffects:FindFirstChild("KO")
        if ko and ko.Value == true then
            return false
        end
    end

    return true
end

local function Gradient(parent, c1, c2, rot)
    local g = Instance.new("UIGradient", parent)
    g.Color = ColorSequence.new(c1, c2)
    g.Rotation = rot or 0
    return g
end

local function CreateToggle(parent, text, defaultState, callback)
    local btn = Instance.new("TextButton", parent)
    btn.Size = UDim2.new(1, 0, 0, 40)
    btn.Text = ""
    btn.AutoButtonColor = false
    btn.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    local btnGrad = Gradient(btn, THEME.panel, THEME.panel2, 90)
    Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 10)
    local stroke = Instance.new("UIStroke", btn)
    stroke.Color = THEME.stroke
    stroke.Thickness = 1.2

    local knob = Instance.new("Frame", btn)
    knob.Size = UDim2.new(0, 18, 0, 18)
    knob.Position = UDim2.new(0, 8, 0.5, -9)
    knob.BackgroundColor3 = THEME.muted
    Instance.new("UICorner", knob).CornerRadius = UDim.new(1, 0)

    local label = Instance.new("TextLabel", btn)
    label.BackgroundTransparency = 1
    label.Position = UDim2.new(0, 34, 0, 0)
    label.Size = UDim2.new(1, -104, 1, 0)
    label.Font = Enum.Font.GothamSemibold
    label.TextSize = 12
    label.Text = text
    label.TextColor3 = THEME.text
    label.TextXAlignment = Enum.TextXAlignment.Left

    local stateLabel = Instance.new("TextLabel", btn)
    stateLabel.BackgroundTransparency = 1
    stateLabel.Position = UDim2.new(1, -62, 0, 0)
    stateLabel.Size = UDim2.new(0, 30, 1, 0)
    stateLabel.Font = Enum.Font.GothamBold
    stateLabel.TextSize = 10
    stateLabel.TextXAlignment = Enum.TextXAlignment.Right

    local active = defaultState or false

    local function paint()
        if active then
            btnGrad.Color = ColorSequence.new(THEME.accent1, THEME.accent2)
            stroke.Color = THEME.accent2
            label.TextColor3 = Color3.fromRGB(255, 255, 255)
            stateLabel.Text = "ON"
            stateLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
            knob.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
            TweenService:Create(knob, TweenInfo.new(0.15), {Position = UDim2.new(1, -26, 0.5, -9)}):Play()
        else
            btnGrad.Color = ColorSequence.new(THEME.panel, THEME.panel2)
            stroke.Color = THEME.stroke
            label.TextColor3 = THEME.text
            stateLabel.Text = "OFF"
            stateLabel.TextColor3 = THEME.muted
            knob.BackgroundColor3 = THEME.muted
            TweenService:Create(knob, TweenInfo.new(0.15), {Position = UDim2.new(0, 8, 0.5, -9)}):Play()
        end
    end
    paint()

    btn.MouseButton1Click:Connect(function()
        active = not active
        paint()
        callback(active)
    end)
    return btn
end

local function GetTargetAimPosition(character)
    local hrp = character and character:FindFirstChild("HumanoidRootPart")
    if hrp then
        return hrp.Position + Vector3.new(0, 0.75, 0)
    end
    return nil
end

local function GetClosestToCenter()
    local cam = Workspace.CurrentCamera
    local closestP = nil
    local shortestDist = Settings.Aimbot.FOV
    local center = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)

    for _, player in ipairs(Players:GetPlayers()) do
        if IsValidTarget(player) then
            local aimPos = GetTargetAimPosition(player.Character)
            if aimPos then
                local pos, onScreen = cam:WorldToViewportPoint(aimPos)
                if onScreen then
                    local magnitude = (Vector2.new(pos.X, pos.Y) - center).Magnitude
                    if magnitude < shortestDist then
                        shortestDist = magnitude
                        closestP = player
                    end
                end
            end
        end
    end
    return closestP
end

-- // RAYCAST HOOK (Fixed argument patterns to prevent crashes)
local HookOK = false
pcall(function()
    local Old
    Old = hookmetamethod(game, "__namecall", newcclosure(function(self, ...)
        if not checkcaller() and HasToolEquipped() then
            local m = getnamecallmethod()
            if m == "Raycast" or m == "FindPartOnRayWithIgnoreList" or m == "FindPartOnRay" or m == "FindPartOnRayWithWhitelist" then
                local focus = GetRayFocusPart()
                if focus then
                    local args = table.pack(...)
                    local origin
                    if m == "Raycast" then
                        origin = args[1]
                    elseif typeof(args[1]) == "Ray" then
                        origin = args[1].Origin
                    end
                    if typeof(origin) == "Vector3" then
                        local newDir = (focus.Position - origin).Unit
                        if m == "Raycast" then
                            local mag = (typeof(args[2]) == "Vector3" and args[2].Magnitude > 0) and args[2].Magnitude or 1000
                            args[2] = newDir * mag
                        else
                            args[1] = Ray.new(origin, newDir * 1000)
                        end
                        RayRedirects = RayRedirects + 1
                        return Old(self, table.unpack(args, 1, args.n))
                    end
                end
            end
        end
        return Old(self, ...)
    end))
    HookOK = true
end)

-- // MOUSEPOS HOOK
local MouseHookOK = false
pcall(function()
    local old
    old = hookmetamethod(game, "__namecall", newcclosure(function(self, ...)
        if not checkcaller() and getnamecallmethod() == "FireServer" and HasToolEquipped() then
            local isRemote = false
            pcall(function() isRemote = self:IsA("RemoteEvent") end)
            if isRemote then
                local Arguments = table.pack(...)
                local a1 = tostring(Arguments[1]):lower()
                if a1 == "updatemousepos" or a1 == "mouse" or a1 == "mousepos" or a1 == "updatemouse" then
                    local focus = GetMouseFocusPart()
                    if focus then
                        pcall(function()
                            Arguments[2] = focus.Position + Vector3.new(
                                focus.Velocity.X * BulletTP.Prediction.Horizontal,
                                focus.Velocity.Y * BulletTP.Prediction.Vertical,
                                focus.Velocity.Z * BulletTP.Prediction.Horizontal
                            )
                        end)
                        MouseRedirects = MouseRedirects + 1
                        return old(self, table.unpack(Arguments, 1, Arguments.n))
                    end
                end
            end
        end
        return old(self, ...)
    end))
    MouseHookOK = true
end)

local function GetClosestPlayer()
    local ClosestPlayer = nil
    local ShortestDistance = math.huge
    local cam = Workspace.CurrentCamera
    for _, Player in pairs(Players:GetPlayers()) do
        if Player ~= LocalPlayer and Player.Character and IsValidTarget(Player) then
            for _, PlayerPart in ipairs(Player.Character:GetChildren()) do
                if PlayerPart:IsA("BasePart") and PlayerPart.Transparency ~= 1 then
                    local PlayerScreenPosition = cam:WorldToViewportPoint(PlayerPart.Position)
                    local PlayerDirection = (PlayerPart.Position - cam.CFrame.Position).Unit
                    local DotProduct = cam.CFrame.LookVector:Dot(PlayerDirection)
                    if DotProduct > 0 then
                        local MagnitudeDistance = (Vector2.new(PlayerScreenPosition.X, PlayerScreenPosition.Y) - FOVCircle.Position).Magnitude
                        if MagnitudeDistance < ShortestDistance and MagnitudeDistance <= FOVCircle.Radius then
                            ClosestPlayer = Player
                            ShortestDistance = MagnitudeDistance
                        end
                    end
                end
            end
        end
    end
    return ClosestPlayer
end

local savedGrips = {}
RunService.Heartbeat:Connect(function()
    local char = LocalPlayer.Character
    if not char then return end
    local wantGrip = BulletTP.Enabled and TargetedPlayerHumanoidRootPart ~= nil and TargetedPlayerHumanoidRootPart.Parent ~= nil
    local myHRP = char:FindFirstChild("HumanoidRootPart")
    for _, tool in ipairs(char:GetChildren()) do
        if tool:IsA("Tool") then
            if wantGrip and myHRP then
                if savedGrips[tool] == nil then
                    savedGrips[tool] = tool.GripPos
                end
                pcall(function()
                    local p = myHRP.CFrame:PointToObjectSpace(TargetedPlayerHumanoidRootPart.CFrame.Position)
                    tool.GripPos = Vector3.new(-p.Z, 0, p.X)
                    GripFrames = GripFrames + 1
                end)
            elseif savedGrips[tool] ~= nil then
                pcall(function() tool.GripPos = savedGrips[tool] end)
                savedGrips[tool] = nil
            end
        end
    end
end)
